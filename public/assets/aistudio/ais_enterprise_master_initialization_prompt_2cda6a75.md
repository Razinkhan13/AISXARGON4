```sql

CREATE TABLE tenants (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    company_name TEXT NOT NULL,

    bin_number TEXT,
    contact_phone TEXT NOT NULL,

    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE core_entities (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    tenant_id UUID REFERENCES tenants(id) ON DELETE CASCADE,
    entity_type TEXT NOT NULL, -- 'APARTMENT_UNIT', 'SANDBOX_SITE', or 'PRODUCT_SKU'
    name TEXT NOT NULL,
    base_price NUMERIC(14, 2) NOT NULL,
    custom_attributes JSONB NOT NULL DEFAULT '{}',
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE operational_transactions (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    tenant_id UUID REFERENCES tenants(id) ON DELETE CASCADE,
    transaction_type TEXT NOT NULL, -- 'REAL_ESTATE_LEAD', 'SANDBOX_ESCROW', or 'ECOMMERCE_COD'
    customer_name TEXT NOT NULL,
    customer_phone TEXT NOT NULL,
    status TEXT NOT NULL DEFAULT 'AWAITING_HUMAN_LOCK',
    total_amount NUMERIC(14, 2) DEFAULT 0.00,
    payload_data JSONB NOT NULL DEFAULT '{}',
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE tenant_scenarios (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    tenant_id UUID REFERENCES tenants(id) ON DELETE CASCADE,
    trigger_event TEXT NOT NULL,
    condition_rules JSONB NOT NULL,
    webhook_url TEXT NOT NULL,
    is_active BOOLEAN DEFAULT TRUE
);

CREATE INDEX idx_entities_custom ON core_entities USING gin (custom_attributes);
CREATE INDEX idx_tx_payload ON operational_transactions USING gin (payload_data);
```
